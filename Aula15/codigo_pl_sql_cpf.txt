create or replace function valida_cpf(cpf varchar2) return varchar2 is
m_total number default 0;
m_digito number default 0;

begin
for i in 1..9 loop
m_total := m_total + substr (cpf, i, 1) * (11 - i);
end loop;

m_digito := 11 - mod (m_total, 11);

if m_digito > 9 then
m_digito := 0;
end if;

if m_digito != substr (cpf, 10, 1) then
return 'I';
end if;

m_digito := 0;
m_total := 0;

for i in 1..10 loop
m_total := m_total + substr (cpf, i, 1) * (12 - i);
end loop;

m_digito := 11 - mod (m_total, 11);

if m_digito > 9 then
m_digito := 0;
end if;

if m_digito != substr (cpf, 11, 1) then
return 'CPF Inválido!';
end if;

return 'CPF Válido';
end;

begin
dbms_output.put_line(valida_cpf('11122233396'));
end;
