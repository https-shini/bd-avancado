-- ============================================================
-- 1. BLOCO DE LIMPEZA DO AMBIENTE (IDEMPOTÊNCIA)
-- Conceito: Processamento de Transações / Boas Práticas (DELETE/DROP)
-- Esta seção garante que o script possa ser executado múltiplas vezes.
-- ============================================================

-- Remove o TRIGGER primeiro, pois ele depende da tabela tbCliente.
BEGIN
    EXECUTE IMMEDIATE 'DROP TRIGGER trg_auditar_cliente';
EXCEPTION
    WHEN OTHERS THEN NULL; -- Ignora o erro se o trigger não existir.
END;
/

-- Remove a tabela de Log (tbLogCliente).
BEGIN
    EXECUTE IMMEDIATE 'DROP TABLE tbLogCliente CASCADE CONSTRAINTS';
EXCEPTION
    WHEN OTHERS THEN NULL;
END;
/

-- Remove a tabela de Usuários (tbUsuario).
BEGIN
    EXECUTE IMMEDIATE 'DROP TABLE tbUsuario CASCADE CONSTRAINTS';
EXCEPTION
    WHEN OTHERS THEN NULL;
END;
/

-- Remove a tabela de Clientes (tbCliente).
BEGIN
    EXECUTE IMMEDIATE 'DROP TABLE tbCliente CASCADE CONSTRAINTS';
EXCEPTION
    WHEN OTHERS THEN NULL;
END;
/

-- ============================================================
-- 2. CRIAÇÃO DAS TABELAS BASE (DDL)
-- Conceito: SQL Avançado / Geração de Identidade
-- ============================================================

-- Tabela de Clientes.
CREATE TABLE tbCliente(
    -- Chave Primária com auto-incremento (IDENTITY).
    idCliente NUMBER GENERATED ALWAYS AS IDENTITY primary key,
    nomeCliente varchar2(50),
    dataNasc date
);

-- Tabela de Usuários (simula quem está logado).
CREATE TABLE tbUsuario(
    idUsuario NUMBER GENERATED ALWAYS AS IDENTITY primary key,
    nomeUsuario varchar2(50),
    senhaUsuario varchar2(40)
);

-- Tabela de Log para registrar as operações de auditoria.
CREATE TABLE tbLogCliente(
    idLogCli NUMBER GENERATED ALWAYS AS IDENTITY primary key,
    idCliente number, -- Qual cliente foi afetado.
    idUsuario number, -- Quem fez a operação (simulado).
    rotina varchar2(40), -- Tipo de operação (INSERT, UPDATE, DELETE).
    dataOperacao DATE DEFAULT SYSDATE
);

-- ============================================================
-- 3. CRIAÇÃO DA TRIGGER DE AUDITORIA
-- Conceito: Triggers (Definição, Codificação e Restrições) / PL/SQL
-- ============================================================
CREATE OR REPLACE TRIGGER trg_auditar_cliente
-- Dispara DEPOIS da execução de qualquer comando DML na tabela tbCliente.
AFTER INSERT OR UPDATE OR DELETE ON tbCliente
-- O trigger é disparado para cada linha afetada.
FOR EACH ROW
DECLARE
    -- Variáveis locais para armazenar o tipo de operação e o ID a ser logado.
    v_rotina VARCHAR2(40);
    v_id_cliente NUMBER;
    -- Constante para simular o ID do usuário logado (AdminTest = 1).
    v_id_usuario CONSTANT NUMBER := 1;
BEGIN
    -- Estrutura de controle PL/SQL para identificar a operação (IF-ELSIF-ELSE).
    IF INSERTING THEN
        v_rotina := 'INSERT';
        -- Para INSERT, usa-se :NEW para obter o ID recém-criado.
        v_id_cliente := :NEW.idCliente;
    ELSIF UPDATING THEN
        v_rotina := 'UPDATE';
        -- Para UPDATE, usa-se :OLD para obter o ID existente (antes da mudança).
        v_id_cliente := :OLD.idCliente;
    ELSIF DELETING THEN
        v_rotina := 'DELETE';
        -- Para DELETE, usa-se :OLD para obter o ID que será removido.
        v_id_cliente := :OLD.idCliente;
    END IF;

    -- Comando INSERT dentro do Trigger para registrar a auditoria.
    INSERT INTO tbLogCliente (idCliente, idUsuario, rotina)
    VALUES (
        v_id_cliente, -- ID do cliente auditado.
        v_id_usuario, -- ID do usuário (simulado).
        v_rotina      -- Tipo de operação.
    );
END;
/
