before update on tbConta

    v_saldo float;

		raise_application_error(-20001,'Saldo Insuficiente');

    end if;
end;

    idUsuario number GENERATED ALWAYS AS IDENTITY primary key,

    senhaUsuario varchar2(40)
);

create table tbLogCliente(
    idLogCli number  GENERATED ALWAYS AS IDENTITY primary key,
    idCliente number,
    idUsuario number,
    rotina varchar2(40), 
    dataOperacao DATE DEFAULT SYSDATE       
);

CREATE OR REPLACE TRIGGER trg_log_cliente
AFTER INSERT OR UPDATE OR DELETE ON tbCliente
FOR EACH ROW
DECLARE
    v_operacao VARCHAR2(10);
BEGIN
    IF INSERTING THEN
        v_operacao := 'INSERT';
    ELSIF UPDATING THEN
        v_operacao := 'UPDATE';
    ELSIF DELETING THEN
        v_operacao := 'DELETE';
    END IF;

    INSERT INTO tbLogCliente (idCliente, rotina, dataOperacao)
    VALUES (
        :new.idCliente,
        v_operacao,
        SYSDATE
    );
END;
