CREATE TABLE EMPREGADOS (
    ID NUMBER PRIMARY KEY,
    NOME VARCHAR2(100),
    SALARIO NUMBER(10, 2),
    DATA_ATUALIZACAO DATE
);

CREATE TABLE HISTORICO_EMPREGADOS (
    ID NUMBER,
    NOME_ANTIGO VARCHAR2(100),
    SALARIO_ANTIGO NUMBER(10, 2),
    DATA_ALTERACAO DATE
);

CREATE OR REPLACE TRIGGER TRIGGER_ALTERACAO_SALARIO
BEFORE UPDATE OF SALARIO ON EMPREGADOS
FOR EACH ROW
BEGIN
    INSERT INTO HISTORICO_EMPREGADOS (ID, NOME_ANTIGO, SALARIO_ANTIGO, DATA_ALTERACAO)
    VALUES (:OLD.ID, :OLD.NOME, :OLD.SALARIO, SYSDATE);
END;
