CREATE TABLE ALUNO (    
	IDALUNO NUMBER PRIMARY KEY,
	NOME VARCHAR2(100),
	DATA_NASCIMENTO DATE,
	EMAIL VARCHAR2(100),
	TELEFONE VARCHAR2(15)
);

CREATE OR REPLACE TRIGGER trg_restricao_quarta
BEFORE INSERT OR UPDATE OR DELETE
ON ALUNO
FOR EACH ROW
DECLARE
  v_dia_semana VARCHAR2(10);
BEGIN
  -- Obtém o nome do dia da semana
  v_dia_semana := TO_CHAR(SYSDATE, 'DAY');

  -- Verifica se é quarta-feira
  IF TRIM(UPPER(v_dia_semana)) = 'SUNDAY' THEN
    RAISE_APPLICATION_ERROR(
      -20001, 
      'Operação não permitida HOJE.'
    );
  END IF;
END;
